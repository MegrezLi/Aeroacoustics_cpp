cmake_minimum_required(VERSION 3.20)
project(Aeroacoustics_cpp VERSION 0.2.0 LANGUAGES CXX)
option(AEROACOUSTICS_USE_MKL "Use real oneMKL VML and BLAS in TNO quadrature" OFF)
option(AEROACOUSTICS_BUILD_EXAMPLES "Build standalone section example" ON)
option(AEROACOUSTICS_BUILD_TESTS "Build differential-test probes" OFF)
option(AEROACOUSTICS_BUILD_TURBINE "Build the standalone turbine solver" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(aeroacoustics STATIC src/kernels.cpp src/aeroacoustics.cpp src/driver.cpp src/c_api.cpp)
target_include_directories(aeroacoustics PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include/aeroacoustics>)
set_target_properties(aeroacoustics PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(AEROACOUSTICS_USE_MKL)
  set(MKL_THREADING sequential)
  set(MKL_INTERFACE lp64)
  if(MINGW)
    # SDL avoids the MSVC-only interface objects in the component import libs.
    set(MKL_LINK sdl)
  endif()
  find_package(MKL CONFIG REQUIRED)
  target_compile_definitions(aeroacoustics PUBLIC AERO_USE_MKL)
  target_link_libraries(aeroacoustics PUBLIC MKL::MKL)
endif()
if(AEROACOUSTICS_BUILD_TESTS)
  add_executable(auxiliary_probe tests/auxiliary_probe.cpp)
  target_link_libraries(auxiliary_probe PRIVATE aeroacoustics)
endif()
add_library(aeroacoustics_shared SHARED src/kernels.cpp src/aeroacoustics.cpp src/driver.cpp src/c_api.cpp)
target_include_directories(aeroacoustics_shared PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(AEROACOUSTICS_USE_MKL)
  target_compile_definitions(aeroacoustics_shared PUBLIC AERO_USE_MKL)
  target_link_libraries(aeroacoustics_shared PUBLIC MKL::MKL)
endif()
if(AEROACOUSTICS_BUILD_EXAMPLES)
  add_executable(aeroacoustics_example src/main.cpp)
  target_link_libraries(aeroacoustics_example PRIVATE aeroacoustics)
endif()
if(AEROACOUSTICS_BUILD_TURBINE)
  add_library(turbine_modules STATIC src/turbine/input.cpp src/turbine/bem.cpp src/turbine/unsteady.cpp src/turbine/structure.cpp src/turbine/mesh.cpp src/turbine/rotor.cpp src/turbine/solver.cpp)
  target_include_directories(turbine_modules PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_link_libraries(turbine_modules PUBLIC aeroacoustics)
  if(AEROACOUSTICS_BUILD_TESTS)
    foreach(probe module structure mesh solver bem)
      add_executable(turbine_${probe}_probe tests/turbine_${probe}_probe.cpp)
      target_link_libraries(turbine_${probe}_probe PRIVATE turbine_modules)
    endforeach()
  endif()
  add_executable(aeroacoustics_turbine src/turbine/main.cpp)
  target_link_libraries(aeroacoustics_turbine PRIVATE turbine_modules)
endif()
